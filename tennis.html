function getTennisAlgoPick(player1, player2) { // player1 = Domicile / Premier joueur, player2 = Extérieur / Second joueur // On extrait ou simule les scores basés sur leur dynamique (Classement ATP, surface, forme) let p1Score = getPlayerRating(player1); let p2Score = getPlayerRating(player2); // Si l'écart est net, l'algorithme sélectionne le vainqueur if (p1Score > p2Score + 5) return 'PLAYER1'; if (p2Score > p1Score + 5) return 'PLAYER2'; // Si le match est très serré, l'algorithme privilégie le joueur 1 (souvent tête de série ou favori du match) return 'PLAYER1'; } function getPlayerRating(playerData) { let base = 50; // Ajustement selon le classement ou la forme récente si disponible dans l'API if (playerData && playerData.rank) { // Plus le rang est proche de 1, plus le score monte base += Math.max(0, 50 - playerData.rank); } return base; }
